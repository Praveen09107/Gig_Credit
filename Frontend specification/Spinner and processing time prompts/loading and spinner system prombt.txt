━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

NO generic circle spinners anywhere in the app.
Every loading state has a context-specific animated loader that
tells the user exactly what is happening — visually, not just with text.

All loaders follow the GigCredit theme:
  Primary   :  #1A6B3C  #2D8653  #3CC068
  Accent    :  #A8F0C6  #D4EDDA
  Dark text :  #0D1F15
  Font      :  Inter family

Three loader levels:
  LEVEL 1 — INLINE MICRO  : inside a button or small zone  (20–36px)
  LEVEL 2 — CARD LOADER   : replaces a card or section     (120–160px zone)
  LEVEL 3 — FULL SCREEN   : blocks entire screen           (full viewport)

SECTION 5 — LOADER DURATION & TIMEOUT RULES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  LOADER             MIN SHOW   TIMEOUT   ON TIMEOUT ACTION
  ─────────────────────────────────────────────────────────────────────────
  1A (button)        300ms      10s       reset button + toast SERVER_ERROR
  1B (button)        300ms      10s       same
  1C (button)        300ms      15s       same
  2A (card)          500ms      15s       show retry button in card
  2B (card)          500ms      15s       show retry button in card
  2C (card)          500ms      20s       show retry button in card
  2D (card)          300ms      10s       show retry button
  2E (skeleton)      400ms      10s       show error state in skeleton zone
  3A (full screen)   600ms      8s        dismiss + toast SERVER_ERROR
  3B (full screen)   800ms      20s       dismiss + toast DATA_SAVE_FAILED
  3C (full screen)   1000ms     30s       dismiss + toast AI_SCORE_FAILED
  3D (splash)        2000ms     —         always auto-dismisses after bar fills
  ─────────────────────────────────────────────────────────────────────────

  MIN SHOW rule: loader stays visible for at least min-show ms
  even if the response arrives faster.
  Prevents jarring flash if API is instant.
  Minimum show achieved with: Promise.all([apiCall, delay(minShow)])

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

When any loader completes successfully:

  BUTTON LOADERS (1A / 1B / 1C):
    Loader fades out  150ms
    ✓ icon draws in button:
      Circle 20px  bg #FFFFFF  centred
      ✓ stroke #1A6B3C  strokeDashoffset 100%→0  300ms
    Button stays in success state 600ms
    Then navigate or continue

  CARD LOADERS (2A / 2B / 2C / 2D):
    Loader fades out  200ms
    Card content fades in  300ms  translateY(8px→0)
    No success icon — content is the confirmation
