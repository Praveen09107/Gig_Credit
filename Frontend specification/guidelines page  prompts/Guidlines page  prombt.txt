━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Screen enter:
  opacity 0→1  translateY(24px→0)  400ms ease

Hero band:
  Gradient slow shift  8s ease-in-out infinite alternate

Step tracker pills:
  Active: bg transition 200ms  scale spring on press
  Completed: bg + prefix ✓ update as user scrolls

Step cards on mount:
  Stagger 60ms × stepIndex  opacity 0→1  translateY(24px→0)  500ms

Type A rows on mount:
  Stagger 40ms × inputIndex within card  opacity 0→1  delay after card enters

Type B droplet expand:
  height auto  350ms  ease[0.34,1.56,0.64,1]
  Content stagger (80ms each):
    whatIsThis → howToGet steps → resource buttons → tip
  Chevron rotates 180°  300ms ease

Type B droplet collapse:
  Reverse  250ms ease-in

Resource button press:
  scale 0.96 spring  haptic 8ms
  If link empty: toast slides up from bottom  "Link coming soon"  3s dismiss

All button presses:
  scale spring  stiffness 400  damping 22

Reduce motion:
  All translateY/translateX/scale skipped
  Droplet: instant open/close (no height animation)
  Values update immediately
